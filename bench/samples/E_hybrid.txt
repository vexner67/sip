### sum_even_squares
fn sum_even_squares(nums [int]) int = sum(n * n for n in nums if n % 2 == 0)

### shapes_match
type Shape = Circle(r float) | Rect(w float, h float) | Triangle(b float, h float)

fn area(s Shape) float = match s
    Circle(r) -> pi * r * r
    Rect(w, h) -> w * h
    Triangle(b, h) -> 0.5 * b * h

### csv_groupby
fn avg_salary_by_dept(path str) {str: float}
    groups = csv.read(path).group_by("dept")
    {dept: avg(float(r.salary) for r in rows) for dept, rows in groups}

### parse_config
DEFAULTS = {host: "localhost", port: 8080, debug: false}

fn load_config(path str) dict
    try DEFAULTS | json.parse(read(path))
    catch NotFound -> DEFAULTS
    catch ParseError e -> fail "bad config: {e}"

### concurrent_fetch
async fn fetch_all(urls [str]) [int] = [r.status for r in await all(http.get(u) for u in urls)]

### binary_search
fn binary_search(items [int], target int) int
    lo, hi = 0, len(items) - 1
    while lo <= hi
        mid = (lo + hi) / 2
        if items[mid] == target then return mid
        if items[mid] < target then lo = mid + 1 else hi = mid - 1
    -1
